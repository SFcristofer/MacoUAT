// ==============================================================================
// SCRIPT 1: PRUEBA DE LECTURA DE PEDIDOS (Para motivos de artículos no surtidos)
// Objetivo: Demostrar que el usuario de integración no tiene permisos (HTTP 403)
// para leer el detalle de los pedidos desde Salesforce.
// ==============================================================================

String token = ZafiroAuthManager.getToken();
Http http = new Http();
HttpRequest req = new HttpRequest();
// Pedido 001-96934 -> AlmacenID 001, PedidoID 96934
req.setEndpoint('callout:IntegracionZafiro/pedidos/001/96934');
req.setMethod('GET');
req.setHeader('Authorization', 'Bearer ' + token);
req.setHeader('Content-Type', 'application/json');

try {
    HttpResponse res = http.send(req);
    System.debug('>>> ESTATUS HTTP: ' + res.getStatusCode());
    System.debug('>>> RESPUESTA ZAFIRO: ' + res.getBody());
} catch(Exception e) {
    System.debug('>>> ERROR: ' + e.getMessage());
}
